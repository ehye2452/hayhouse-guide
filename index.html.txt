<script>
(function () {
  var C = window.CONFIG || {};
  var T = C.text || {};
  var app = document.getElementById("app");

  try {
    document.documentElement.style.setProperty("--accent", C.accentColor || "#C8642B");
    var theme = document.createElement("meta");
    theme.name = "theme-color";
    theme.content = "#FFFFFF";
    document.head.appendChild(theme);
  } catch (e) {}
  document.title = C.stayName || "";

  function el(tag, cls, text) {
    var e = document.createElement(tag);
    if (cls) e.className = cls;
    if (text !== undefined && text !== null) e.textContent = text;
    return e;
  }
  function onlyNumber(s) {
    return String(s || "").replace(/[^0-9+]/g, "");
  }
  function fill(tpl, data) {
    return String(tpl || "").replace(/\{(\w+)\}/g, function (m, k) {
      return data[k] !== undefined ? data[k] : "";
    });
  }

  var toastEl = el("div", "toast");
  toastEl.setAttribute("role", "status");
  toastEl.setAttribute("aria-live", "polite");
  document.body.appendChild(toastEl);
  var toastTimer = null;
  function toast(msg) {
    toastEl.textContent = msg || "";
    toastEl.classList.add("show");
    clearTimeout(toastTimer);
    toastTimer = setTimeout(function () { toastEl.classList.remove("show"); }, 2200);
  }

  function fallbackCopy(text) {
    var ta = document.createElement("textarea");
    ta.value = text;
    ta.setAttribute("readonly", "");
    ta.style.position = "fixed";
    ta.style.top = "-1000px";
    ta.style.opacity = "0";
    document.body.appendChild(ta);
    ta.select();
    try { ta.setSelectionRange(0, text.length); } catch (e) {}
    var ok = false;
    try { ok = document.execCommand("copy"); } catch (e) { ok = false; }
    document.body.removeChild(ta);
    return ok;
  }
  function copyText(text) {
    text = String(text || "");
    if (navigator.clipboard && window.isSecureContext) {
      navigator.clipboard.writeText(text).then(
        function () { toast(T.copied); },
        function () { toast(fallbackCopy(text) ? T.copied : T.copyFailed); }
      );
    } else {
      toast(fallbackCopy(text) ? T.copied : T.copyFailed);
    }
  }

  function button(label, cls, onClick) {
    var b = el("button", "btn" + (cls ? " " + cls : ""), label);
    b.type = "button";
    b.addEventListener("click", onClick);
    return b;
  }
  function linkBtn(label, href, cls, newTab) {
    var a = el("a", "btn" + (cls ? " " + cls : ""), label);
    a.href = href;
    if (newTab) {
      a.target = "_blank";
      a.rel = "noopener";
    }
    return a;
  }
  function info(label, value, valueCls) {
    var w = el("div");
    w.appendChild(el("p", "label", label));
    w.appendChild(el("p", valueCls || "value", value));
    return w;
  }
  function backButton() {
    return linkBtn(T.backToMenu, "#", "btn-outline back");
  }
  function mapUrl(type, q) {
    var enc = encodeURIComponent(q || "");
    if (type === "naver") return "https://map.naver.com/p/search/" + enc;
    if (type === "kakao") return "https://map.kakao.com/link/search/" + enc;
    if (type === "google") return "https://www.google.com/maps/search/?api=1&query=" + enc;
    return "";
  }

  function renderHome() {
    var hero = el("div", "photo-wrap");
    hero.appendChild(el("div", "photo-bg"));
    var ph = el("p", "photo-placeholder", T.photoPlaceholder);
    hero.appendChild(ph);
    if (C.photo) {
      var img = new Image();
      img.className = "photo";
      img.alt = C.photoAlt || "";
      img.onload = function () {
        img.classList.add("loaded");
        ph.style.display = "none";
      };
      img.onerror = function () {
        img.style.display = "none";
      };
      img.src = C.photo;
      hero.appendChild(img);
    }
    app.appendChild(hero);

    app.appendChild(el("h1", "", C.stayName));
    app.appendChild(el("p", "tagline", C.tagline));
    app.appendChild(el("p", "capacity", fill(T.capacityLine, {
      guests: C.maxGuests,
      bedrooms: C.bedrooms,
      bathrooms: C.bathrooms
    })));

    app.appendChild(el("h2", "section-title", T.menuTitle));
    var grid = el("div", "menu-grid");
    grid.appendChild(linkBtn(T.menuCheckin, "#checkin", "menu-btn"));
    grid.appendChild(linkBtn(T.menuWifi, "#wifi", "menu-btn"));
    grid.appendChild(linkBtn(T.menuContact, "#contact", "menu-btn"));
    app.appendChild(grid);
  }

  function renderCheckin() {
    app.appendChild(el("h2", "screen-title", T.checkinTitle));

    var addr = el("div", "card stack");
    addr.appendChild(info(T.labelAddress, C.address));
    addr.appendChild(button(T.copyAddress, "btn-outline", function () { copyText(C.address); }));
    app.appendChild(addr);

    var maps = C.mapApps || [];
    if (maps.length) {
      var mapCard = el("div", "card stack");
      mapCard.appendChild(el("p", "label", T.labelMap));
      for (var i = 0; i < maps.length; i++) {
        var url = mapUrl(maps[i].type, C.mapSearchAddress || C.address);
        if (url) mapCard.appendChild(linkBtn(maps[i].label, url, "", true));
      }
      app.appendChild(mapCard);
    }

    var time = el("div", "card");
    var row = el("div", "row");
    row.appendChild(info(T.labelCheckin, C.checkinTime));
    row.appendChild(info(T.labelCheckout, C.checkoutTime));
    time.appendChild(row);
    app.appendChild(time);

    var door = el("div", "card");
    door.appendChild(el("p", "label", T.labelDoor));
    door.appendChild(el("p", "text", C.doorGuide));
    door.appendChild(el("p", "notice", C.doorCodeNotice));
    app.appendChild(door);

    app.appendChild(backButton());
  }

  function renderWifi() {
    app.appendChild(el("h2", "screen-title", T.wifiTitle));

    var card = el("div", "card");
    var nameBox = el("div", "stack");
    nameBox.appendChild(info(T.labelWifiName, C.wifiName, "code-value"));
    nameBox.appendChild(button(T.copyWifiName, "btn-outline", function () { copyText(C.wifiName); }));
    card.appendChild(nameBox);
    card.appendChild(el("div", "gap"));
    var pwBox = el("div", "stack");
    pwBox.appendChild(info(T.labelWifiPassword, C.wifiPassword, "code-value"));
    pwBox.appendChild(button(T.copyWifiPassword, "", function () { copyText(C.wifiPassword); }));
    card.appendChild(pwBox);
    app.appendChild(card);

    app.appendChild(backButton());
  }

  function renderContact() {
    app.appendChild(el("h2", "screen-title", T.contactTitle));
    app.appendChild(el("p", "text", T.contactIntro));
    app.appendChild(el("div", "gap"));

    var card = el("div", "card stack");
    card.appendChild(info(T.labelHost, C.hostName));
    var num = onlyNumber(C.phone);
    if (num) {
      card.appendChild(linkBtn(T.callButton, "tel:" + num));
      card.appendChild(linkBtn(T.smsButton, "sms:" + num, "btn-outline"));
    }
    if (C.kakao) {
      card.appendChild(linkBtn(T.kakaoButton, C.kakao, "btn-outline", true));
    }
    if (C.whatsapp) {
      card.appendChild(linkBtn(T.whatsappButton, "https://wa.me/" + String(C.whatsapp).replace(/[^0-9]/g, ""), "btn-outline", true));
    }
    if (C.line) {
